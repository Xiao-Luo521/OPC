// ===== 董事会AI会议 - 飞书多维表格云函数代理 ===== // 运行时：Node.js 18.15 | 入口：index.main // 环境变量：FEISHU_APP_ID, FEISHU_APP_SECRET, FEISHU_BITABLE_APP, FEISHU_TABLE_ID const https = require('https'); const FEISHU_APP_ID = process.env.FEISHU_APP_ID || ''; const FEISHU_APP_SECRET = process.env.FEISHU_APP_SECRET || ''; const BITABLE_APP = process.env.FEISHU_BITABLE_APP || ''; const TABLE_ID = process.env.FEISHU_TABLE_ID || ''; // ===== 飞书 Token 缓存 ===== let _token = ''; let _tokenExp = 0; async function getToken() { if (_token && Date.now() < _tokenExp - 60000) return _token; const data = await feishuRequest('POST', '/open-apis/auth/v3/tenant_access_token/internal', { app_id: FEISHU_APP_ID, app_secret: FEISHU_APP_SECRET }); if (data.code === 0 && data.tenant_access_token) { _token = data.tenant_access_token; _tokenExp = Date.now() + (data.expire || 7200) * 1000; return _token; } throw new Error('Token获取失败: ' + JSON.stringify(data)); } // ===== 飞书原始请求 ===== function feishuRequest(method, path, body, token) { return new Promise((resolve, reject) => { const options = { hostname: 'open.feishu.cn', path: path, method: method, headers: { 'Content-Type': 'application/json' } }; if (token) options.headers['Authorization'] = 'Bearer ' + token; const req = https.request(options, (res) => { let data = ''; res.on('data', chunk => data += chunk); res.on('end', () => { try { resolve(JSON.parse(data)); } catch(e) { reject(new Error('JSON解析失败: ' + data.substring(0, 200))); } }); }); req.on('error', reject); req.setTimeout(15000, () => { req.destroy(); reject(new Error('请求超时')); }); if (body) req.write(JSON.stringify(body)); req.end(); }); } // ===== 字段名 → 字段ID 映射 ===== const FIELD_MAP = { '议题': 'fld0Z7y3H2', '沟通轮次': 'fldtPhXsY0', '高管': 'fldI8VuVO5', '对话内容': 'fldhzZSmLp', '问题': 'fldslwsMJE', '解决方案': 'fldKqut30l' }; // ===== 写入操作 ===== async function handleWrite(records) { const token = await getToken(); const url = `/open-apis/bitable/v1/apps/${BITABLE_APP}/tables/${TABLE_ID}/records/batch_create`; const converted = records.map(rec => { const fields = rec.fields || rec; const convertedFields = {}; for (const [key, value] of Object.entries(fields)) { const fieldId = FIELD_MAP[key] || key; convertedFields[fieldId] = value; } return { fields: convertedFields }; }); const result = await feishuRequest('POST', url, { records: converted }, token); if (result.code !== 0) { console.error('写入失败:', result.msg, result); return { code: result.code, msg: result.msg || '写入失败' }; } console.log('成功写入', converted.length, '条记录'); return { code: 0, msg: 'ok', data: { count: converted.length } }; } // ===== 读取操作 ===== async function handleRead(agenda) { const token = await getToken(); const all = []; let pageToken = ''; let page = 0; try { do { let url = `/open-apis/bitable/v1/apps/${BITABLE_APP}/tables/${TABLE_ID}/records?page_size=100`; if (pageToken) url += '&page_token=' + encodeURIComponent(pageToken); const d = await feishuRequest('GET', url, null, token); if (d.code === 0 && d.data && d.data.items) { const reverseMap = {}; for (const [name, id] of Object.entries(FIELD_MAP)) { reverseMap[id] = name; } const items = d.data.items.map(item => { const newFields = {}; for (const [key, value] of Object.entries(item.fields || {})) { const name = reverseMap[key] || key; newFields[name] = value; } return { ...item, fields: newFields }; }); all.push(...items); pageToken = d.data.has_more ? (d.data.page_token || '') : ''; page++; } else { console.error('读取失败:', d.msg); break; } if (page > 20) break; } while (pageToken); } catch(e) { console.error('读取异常:', e.message); return { code: -1, msg: e.message }; } console.log('读取到', all.length, '条记录'); return { code: 0, data: { records: all, total: all.length } }; } // ===== 构建CORS响应 ===== function corsResponse(statusCode, body) { return { isBase64Encoded: false, statusCode: statusCode, headers: { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, GET, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, Authorization, Accept, X-Requested-With', 'Access-Control-Max-Age': '86400', 'Content-Type': 'application/json; charset=utf-8' }, body: typeof body === 'string' ? body : JSON.stringify(body) }; } // ===== 主入口 ===== exports.main = async (event, context) => { // 处理 CORS 预检请求 const method = (event.httpMethod || event.requestContext?.http?.method || '').toUpperCase(); if (method === 'OPTIONS') { return corsResponse(200, ''); } try { let body; if (typeof event.body === 'string') { try { body = JSON.parse(event.body); } catch(e) { body = {}; } } else if (event.body && typeof event.body === 'object') { body = event.body; } else { // 某些网关直接把参数放在 event 顶层 body = event; } const action = body.action; console.log('收到请求, action:', action, ', method:', method); let result; if (action === 'write') { result = await handleWrite(body.records || []); } else if (action === 'read') { result = await handleRead(body.agenda || null); } else { result = { code: -1, msg: '未知action: ' + action }; } return corsResponse(200, result); } catch (e) { console.error('云函数异常:', e); return corsResponse(200, { code: -1, msg: e.message }); } };
